#!/usr/bin/env python3
import json, csv, re

# Patterns for POSITIVE associations (claims a relationship exists)
POSITIVE_PATTERNS = [
    r"(is |was |are |were )?(significantly |strongly |positively )?(associated|correlated) with",
    r"(elevated|increased|higher|raised) (in |levels in )?(patients with |HFpEF|heart failure)",
    r"(predictor|marker|indicator) (of|for)",
    r"(contributes?|contributed) to",
    r"plays? (a |an )?(important |key |critical |crucial )?role in",
    r"(linked|related) to (the )?(pathogenesis|development|progression)",
    r"(promotes?|induces?|causes?|leads? to)",
    r"risk factor for",
]

# Patterns for NO_ASSOCIATION (methods, study design, no claim)
NO_ASSOC_PATTERNS = [
    r"^(we |this study )?(aimed|sought|investigated|examined|evaluated|assessed)",
    r"^(methods|background|objectives?|purpose|aim):",
    r"(patients|subjects|participants) (were |with ).{0,30}(enrolled|included|recruited)",
    r"(retrospective|prospective|cross-sectional|longitudinal) (cohort |observational )?(study|analysis)",
    r"(data|samples) (were |was )(collected|obtained|extracted)",
    r"between (january|february|march|april|may|june|july|august|september|october|november|december|\d{4})",
    r"(inclusion|exclusion) criteria",
    r"we (measured|analyzed|compared|calculated)",
]

# Load PMIDs
pmids = set()
with open("6CVDs/All_HFpEF_PMIDS.csv") as f:
    reader = csv.reader(f)
    next(reader)
    for row in reader:
        if len(row) >= 2:
            pmids.add(row[1])

print(f"Loaded {len(pmids)} PMIDs")

# Scan pubmed.json
sentences = []
with open("../primary_caseolap/data/pubmed.json") as f:
    for i, line in enumerate(f):
        if i % 500000 == 0:
            print(f"Processed {i} lines, found {len(sentences)} sentences...")
        try:
            doc = json.loads(line)
            if str(doc.get('PMID', '')) in pmids:
                abstract = doc.get('Abstract', '')
                if abstract:
                    for sent in re.split(r'(?<=[.!?])\s+', abstract):
                        if 50 < len(sent) < 500:
                            sentences.append({'pmid': doc['PMID'], 'sentence': sent})
        except:
            continue

print(f"Found {len(sentences)} total sentences")

# Compile patterns
pos_compiled = [re.compile(p, re.IGNORECASE) for p in POSITIVE_PATTERNS]
no_assoc_compiled = [re.compile(p, re.IGNORECASE) for p in NO_ASSOC_PATTERNS]

# Match
positives = []
no_assocs = []

for item in sentences:
    sent = item['sentence']
    
    # Check positive
    if len(positives) < 300:
        for p in pos_compiled:
            if p.search(sent):
                positives.append({**item, 'label': 'positive'})
                break
    
    # Check no_association
    if len(no_assocs) < 300:
        for p in no_assoc_compiled:
            if p.search(sent):
                no_assocs.append({**item, 'label': 'no_association'})
                break
    
    if len(positives) >= 300 and len(no_assocs) >= 300:
        break

print(f"Found {len(positives)} positive, {len(no_assocs)} no_association")

# Save
with open('positive_sentences.csv', 'w', newline='') as f:
    writer = csv.DictWriter(f, fieldnames=['pmid', 'sentence', 'label'])
    writer.writeheader()
    writer.writerows(positives)

with open('no_association_sentences.csv', 'w', newline='') as f:
    writer = csv.DictWriter(f, fieldnames=['pmid', 'sentence', 'label'])
    writer.writeheader()
    writer.writerows(no_assocs)

print("Done! Saved positive_sentences.csv and no_association_sentences.csv")
SCRIPT_END

python3 extract_all_classes.py

import csv, json
from pathlib import Path

existing = json.loads(Path('data/labeled.json').read_text())
existing_sents = {item['sentence'] for item in existing}

added = 0
for fname, label in [('data/positive_sentences.csv', 'positive'), 
                      ('data/no_association_sentences.csv', 'no_association')]:
    with open(fname) as f:
        for row in csv.DictReader(f):
            if row['sentence'] not in existing_sents:
                existing.append({'sentence': row['sentence'], 'label': label})
                existing_sents.add(row['sentence'])
                added += 1

Path('data/labeled.json').write_text(json.dumps(existing, indent=2))
