#!/usr/bin/env python3
import json, csv, re

PATTERNS = [
    r"no significant (difference|association|correlation|relationship|effect)",
    r"not (significantly )?associated",
    r"failed to (show|demonstrate|find|detect)",
    r"no (correlation|relationship|evidence|difference)",
    r"did not (show|demonstrate|predict|correlate)",
    r"remains (unclear|to be determined|unknown)",
    r"not (well )?established",
]

# Load PMIDs from column 2, skip header
pmids = set()
with open("6CVDs/All_HFpEF_PMIDS.csv") as f:
    reader = csv.reader(f)
    next(reader)
    for row in reader:
        if len(row) >= 2:
            pmids.add(row[1])

print(f"Loaded {len(pmids)} PMIDs")

# Scan pubmed.json
sentences = []
with open("../primary_caseolap/data/pubmed.json") as f:
    for i, line in enumerate(f):
        if i % 500000 == 0:
            print(f"Processed {i} lines, found {len(sentences)} sentences...")
        try:
            doc = json.loads(line)
            if str(doc.get('PMID', '')) in pmids:
                abstract = doc.get('Abstract', '')
                if abstract:
                    for sent in re.split(r'(?<=[.!?])\s+', abstract):
                        if len(sent) > 50:
                            sentences.append({'pmid': doc['PMID'], 'sentence': sent})
        except:
            continue

print(f"Found {len(sentences)} total sentences")

# Match negative patterns
compiled = [re.compile(p, re.IGNORECASE) for p in PATTERNS]
results = []
for item in sentences:
    for p in compiled:
        if p.search(item['sentence']):
            results.append({**item, 'pattern': p.pattern})
            break
    if len(results) >= 500:
        break

# Save
with open('negative_sentences.csv', 'w', newline='') as f:
    writer = csv.DictWriter(f, fieldnames=['pmid', 'sentence', 'pattern'])
    writer.writeheader()
    writer.writerows(results)

print(f"Saved {len(results)} negative sentences to negative_sentences.csv")
SCRIPT_END
